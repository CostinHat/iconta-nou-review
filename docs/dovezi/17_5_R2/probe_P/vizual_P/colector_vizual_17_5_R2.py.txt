def _acasa(pg):
    pg.goto(BAZA + "/", wait_until="domcontentloaded")
    pg.evaluate("sessionStorage.removeItem('iconta_fereastra_deschisa')")
    pg.reload(wait_until="domcontentloaded")
    pg.locator("button.cab-card:has([data-cheie='firme'])").wait_for()


def _firma(pg, control):
    _acasa(pg)
    pg.locator("button.cab-card:has([data-cheie='firme'])").click()
    pg.locator(TOP + "#opt-existente").click()
    pg.locator(TOP + "#firme-lista button.firme-rand", has_text=F1).first.click()
    t = pg.locator(TOP + control); t.wait_for(state="attached")
    g = t.locator("xpath=ancestor::*[contains(concat(' ',normalize-space(@class),' '),' firme-optiuni ')][1]")
    if g.count() and not g.is_visible():
        pg.locator(TOP + 'button[aria-controls="%s"]' % g.get_attribute("id")).click()
    t.click()


def _setari(pg):
    _acasa(pg)
    b = pg.locator("button.cab-card:has([data-cheie='setari'])")
    if not b.is_visible():
        pg.locator("button[aria-controls='fgc-sef'][aria-expanded='false']").click()
    b.click()

def test_vizual_17_5_R2():
    dest = os.path.join(CAPTURI, "vizual_17_5_R2"); os.makedirs(dest, exist_ok=True)
    rez, erori, erori_js = {}, {}, []
    with sync_playwright() as p:
        b = p.chromium.launch()
        for w, h in REZ:
            ctx = b.new_context(viewport={"width": w, "height": h}); ctx.clock.set_fixed_time(MOMENT_REF)
            ctx.add_init_script(sesiune("dovada.71507@sesiuneab.test")); pg = ctx.new_page()
            pg.on("requestfailed", lambda r, w=w: erori_js.append("%d FAILED %s %s" % (w, r.url[:200], r.failure)) if len(erori_js) < 400 else None)
            pg.on("pageerror", lambda e, w=w: erori_js.append("%d: %s" % (w, str(e)[:300])))
            pg.on("console", lambda m, w=w: erori_js.append("%d console.%s: %s" % (w, m.type, m.text[:300])) if m.type in ("error",) else None)
            for nume, deschide, reper in ECRANE:
                k = "%dx%d/%s" % (w, h, nume)
                pg.close(); pg = ctx.new_page()   # o pagină nouă pe ecran: reîncărcările repetate în aceeași pagină epuizează resursele browserului
                pg.on("requestfailed", lambda r, w=w: erori_js.append("%d FAILED %s %s" % (w, r.url[:200], r.failure)) if len(erori_js) < 400 else None)
                pg.on("pageerror", lambda e, w=w: erori_js.append("%d: %s" % (w, str(e)[:300])))
                try:
                    deschide(pg)
                    pg.locator(TOP + reper).first.wait_for(state="attached", timeout=20000)
                    pg.wait_for_timeout(800)
                    rez[k] = pg.evaluate(MASOARA)
                    pg.screenshot(path=os.path.join(dest, "%dx%d_%s.png" % (w, h, nume)))
                except Exception as e:  # noqa: BLE001 — se consemnează pe ecran, nu oprește restul
                    erori[k] = "%s: %s" % (type(e).__name__, str(e).splitlines()[0] if str(e) else "")
                    pg.screenshot(path=os.path.join(dest, "EROARE_%dx%d_%s.png" % (w, h, nume)))
            ctx.close()
        b.close()
    json.dump({"masuri": rez, "erori": erori, "erori_js": erori_js}, open(os.path.join(dest, "masuri.json"), "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    assert not erori, erori
